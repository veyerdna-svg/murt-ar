Нажмите кнопку, чтобы запустить AR
sceneEl.addEventListener("arReady", () => { startButton.style.display = "none"; statusEl.textContent = "Наведите камеру на знак МУРТ"; }); sceneEl.addEventListener("arError", (event) => { statusEl.textContent = "Ошибка запуска камеры. Попробуйте ещё раз."; console.error("MindAR error:", event); }); startButton.addEventListener("click", async () => { startButton.disabled = true; startButton.textContent = "Запуск..."; statusEl.textContent = "Запрашиваем доступ к камере..."; try { const arSystem = sceneEl.systems["mindar-image-system"]; if (!arSystem) { throw new Error("MindAR system не найден"); } await arSystem.start(); } catch (error) { console.error(error); startButton.disabled = false; startButton.textContent = "Запустить камеру"; statusEl.textContent = "Камера не запустилась: " + (error.name || error.message || "неизвестная ошибка"); } });
